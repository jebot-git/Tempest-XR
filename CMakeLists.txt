cmake_minimum_required(VERSION 3.18)
project(tempest_xr VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(tempest-dome src/domed.cpp)
target_include_directories(tempest-dome PUBLIC src)
target_link_libraries(tempest-dome PRIVATE PkgConfig::CURL Threads::Threads nlohmann_json::nlohmann_json)
find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
option(TEMPEST_OPENXR "Build the native OpenXR backend" ON)
add_library(tempest-game src/game.cpp src/tunnel.cpp src/enemies.cpp src/effects.cpp src/frontend.cpp)
target_include_directories(tempest-game PUBLIC src)
add_library(tempest-music src/music.cpp)
target_include_directories(tempest-music PUBLIC src)
target_link_libraries(tempest-music PRIVATE ${CMAKE_DL_LIBS})
add_library(tempest-records src/records.cpp)
target_link_libraries(tempest-records PUBLIC tempest-game PRIVATE nlohmann_json::nlohmann_json)
add_executable(tempest-xr src/main.cpp src/render.cpp src/render-feedback.cpp src/audio.cpp src/xr.cpp)
target_compile_definitions(tempest-xr PRIVATE TEMPEST_MUSIC_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/music")
target_compile_definitions(tempest-xr PRIVATE TEMPEST_ART_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/art")
target_link_libraries(tempest-xr PRIVATE tempest-records tempest-game tempest-music tempest-dome PkgConfig::SDL2 OpenGL::GL OpenGL::EGL)
target_compile_options(tempest-game PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(tempest-xr PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
if(TEMPEST_OPENXR)
  pkg_check_modules(OPENXR REQUIRED IMPORTED_TARGET openxr)
  find_package(X11 REQUIRED)
  target_compile_definitions(tempest-xr PRIVATE TEMPEST_OPENXR=1 GL_GLEXT_PROTOTYPES)
  target_link_libraries(tempest-xr PRIVATE PkgConfig::OPENXR X11::X11)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(fidelity-test tests/fidelity-test.cpp)
  target_link_libraries(fidelity-test PRIVATE tempest-game)
  add_test(NAME source-frame-fidelity COMMAND fidelity-test)
  add_executable(records-test tests/records-test.cpp)
  target_link_libraries(records-test PRIVATE tempest-records)
  add_test(NAME score-table-and-access-keys COMMAND records-test)
  add_executable(gap-test tests/gap-test.cpp)
  target_link_libraries(gap-test PRIVATE tempest-game)
  add_test(NAME source-gap-regressions COMMAND gap-test)
  add_executable(game-test tests/game-test.cpp)
  target_link_libraries(game-test PRIVATE tempest-game)
  add_test(NAME gameplay COMMAND game-test)
  add_executable(rim-combat-test tests/rim-combat-test.cpp)
  target_link_libraries(rim-combat-test PRIVATE tempest-game)
  add_test(NAME timed-rim-combat COMMAND rim-combat-test)
  add_executable(source-behavior-test tests/source-behavior-test.cpp)
  target_link_libraries(source-behavior-test PRIVATE tempest-game)
  add_test(NAME original-behavior-regressions COMMAND source-behavior-test)
  add_executable(enemy-ai-test tests/enemy-ai-test.cpp)
  target_link_libraries(enemy-ai-test PRIVATE tempest-game)
  add_test(NAME original-enemy-ai COMMAND enemy-ai-test)
  add_executable(tunnel-test tests/tunnel-test.cpp)
  target_link_libraries(tunnel-test PRIVATE tempest-game)
  add_test(NAME original-tunnel COMMAND tunnel-test)
  add_executable(campaign-test tests/campaign-test.cpp)
  target_link_libraries(campaign-test PRIVATE tempest-game)
  add_test(NAME enemies-campaign-regressions COMMAND campaign-test)
  add_executable(music-test tests/music-test.cpp)
  target_link_libraries(music-test PRIVATE tempest-music)
  add_test(NAME soundtrack COMMAND music-test "${CMAKE_CURRENT_SOURCE_DIR}/assets/music")
  add_executable(audio-test tests/audio-test.cpp src/audio.cpp)
  target_link_libraries(audio-test PRIVATE tempest-game tempest-music PkgConfig::SDL2)
  add_test(NAME original-effects-mixer COMMAND audio-test "${CMAKE_CURRENT_SOURCE_DIR}/assets/sfx")
  set_tests_properties(original-effects-mixer PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
  set_tests_properties(soundtrack PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(render-test tests/render-test.cpp src/render.cpp src/render-feedback.cpp)
  target_compile_definitions(render-test PRIVATE TEMPEST_ART_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/art")
  target_link_libraries(render-test PRIVATE tempest-game PkgConfig::SDL2 OpenGL::GL)
  add_test(NAME xr-sky-and-effects COMMAND render-test)
  set_tests_properties(xr-sky-and-effects PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=offscreen")
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME original-asset-integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_assets.py)
  endif()
endif()
install(TARGETS tempest-xr RUNTIME DESTINATION bin)
install(DIRECTORY assets/music/ DESTINATION share/tempest-xr/music)
install(DIRECTORY assets/sfx/ DESTINATION share/tempest-xr/sfx)
install(DIRECTORY assets/art/ DESTINATION share/tempest-xr/art)

configure_file(packaging/org.de_xr.TempestXR.desktop.in org.de_xr.TempestXR.desktop @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.de_xr.TempestXR.desktop" DESTINATION share/applications)
install(FILES assets/icons/org.de_xr.TempestXR.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION share/doc/tempest-xr)
install(DIRECTORY docs/ DESTINATION share/doc/tempest-xr/docs)

foreach(size 16 24 32 48 64 128 256)
  install(FILES assets/icons/${size}x${size}/org.de_xr.TempestXR.png DESTINATION share/icons/hicolor/${size}x${size}/apps)
endforeach()

if(BUILD_TESTING AND TEMPEST_OPENXR)
  add_library(tempest-mock-openxr SHARED tests/mock-openxr.cpp)
  target_link_libraries(tempest-mock-openxr PRIVATE OpenGL::GL PkgConfig::OPENXR)
  if(Python3_Interpreter_FOUND)
    add_test(NAME xr-runtime-failure COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tools/test_xr_failure.py
      --game $<TARGET_FILE:tempest-xr> --runtime $<TARGET_FILE:tempest-mock-openxr>)
  endif()
endif()
